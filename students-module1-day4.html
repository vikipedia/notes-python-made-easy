def median(nums):
    sorted_nums = sorted(nums)
    c = (len(sorted_nums)-1)/2
    n = int(c) # this is integer
    return sorted_nums[n]

def max2(x, y):
    if x > y:
        return x
    else:
        return y

def max3(x, y, z):
    a = max2(x, y)
    return max2(a, z)

67
def max3(x, y, z):
    return max2(max2(x, y), z)
def max3(x, y, z):
    if x > y and x > z:
        return x
    elif y > x and y > z:
        return y
    else:
        return z
