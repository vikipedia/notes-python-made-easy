def is_prime(n):
    return len([f for f in range(1, n+1) if n%f==0])==2

class Prime:
    
    def __init__(self, value):
        if is_prime(value):
            self.value = value
        else:
            raise ValueError("The value passed is not prime")

def computation(n):
    s = 0
    for i in range(n):
        for j in range(n):
            s = s + float(i)*float(j)
    return s

class StopWatch:

    def __init__(self):
        self.starttime = 0
        self.endtime = 0

    def start(self):
        self.starttime = time.time()

    def stop(self):
        self.stoptime = time.time()

    def get_time_elapsed(self):
        return self.stoptime - self.starttime
